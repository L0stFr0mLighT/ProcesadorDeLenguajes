Terminales = { && > < + - ! real cadena entero id write read return = |= 
                let if ( )  { } for int float boolean string void function ; , eof lambda }

NoTerminales = {E E' E1 E1' E2 E2' E3 S S_SIMPLE S_COMPUESTA ASIGFUN A L Q X T INI ACTU C F H B K P}

Axioma = P

Producciones = {
    P -> S P 
    P -> F P
    P -> eof
    
    ////Expresiones:
    E   -> E1 E'
    E'  -> && E1 E'
    E'  -> lambda
    E1  -> E2 E1'
    E1' -> > E2 E1'
    E1' -> < E2 E1'
    E1' -> lambda
    E2  -> E3 E2'
    E2' -> + E3 E2'
    E2' -> - E3 E2' 
    E2' -> lambda
    E3  -> ! E3 
    E3  -> id
    E3  -> ( E )
    E3  -> id ( E )
    E3  -> cadena
    E3  -> entero
    E3  -> real

    ////Sentencias:
    S -> S_SIMPLE
    S -> S_COMPUESTA
    
    S_SIMPLE -> id ASIGFUN
    S_SIMPLE -> write E ;
    S_SIMPLE -> read id ;
    S_SIMPLE -> return X ;
    ASIGFUN  -> A E ;
    ASIGFUN  -> ( L ) ;
    A -> =
    A -> |=
    L -> E Q
    L -> lambda 
    Q -> , E Q 
    Q -> lambda
    X -> E
    X -> lambda

    S_COMPUESTA -> let T id ;
    S_COMPUESTA -> if ( E ) S
    S_COMPUESTA -> for ( INI ; E ; ACTU ) { C }
    T -> int
    T -> float
    T -> boolean
    T -> string
    INI  -> id = E ;
    INI  -> lambda
    ACTU -> id A E ; 
    ACTU -> lambda
    C -> S C 
    C -> lambda


    ////Declaracion de funciones:
    F -> function H id ( B ) { C }
    H -> T
    H -> void
    B -> T id K 
    B -> void
    K -> , T id K 
    K -> lambda
}
